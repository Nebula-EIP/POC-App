42[[nodiscard]] constexpr utils::WrappedColor Hex(std::uint32_t rgb,
43 float opacity = 1.0F) {
44 constexpr std::uint32_t kByte = 0xFFU;
45 constexpr float kMaxAlpha = 255.0F;
46 const float kClamped =
47 opacity < 0.0F ? 0.0F : (opacity > 1.0F ? 1.0F : opacity);
48 return utils::WrappedColor{
49 static_cast<unsigned char>((rgb >> 16U) & kByte),
50 static_cast<unsigned char>((rgb >> 8U) & kByte),
51 static_cast<unsigned char>(rgb & kByte),
52 static_cast<unsigned char>(kClamped * kMaxAlpha + 0.5F),
53 };
54}

62[[nodiscard]] constexpr utils::WrappedColor WithOpacity(
63 utils::WrappedColor color, float opacity) {
64 constexpr float kMaxAlpha = 255.0F;
65 const float kClamped =
66 opacity < 0.0F ? 0.0F : (opacity > 1.0F ? 1.0F : opacity);
67 color.a_ = static_cast<unsigned char>(kClamped * kMaxAlpha + 0.5F);
68 return color;
69}

267[[nodiscard]] constexpr utils::WrappedColor NodeAccent(
268 std::string_view node_name, std::string_view category) {
269 for (const accent::Entry &entry : accent::kNodes) {
270 if (entry.name_ == node_name) {
271 return entry.color_;
272 }
273 }
274 const std::string_view kRoot = category.substr(0, category.find('/'));
275 for (const accent::Entry &entry : accent::kCategories) {
276 if (entry.name_ == kRoot) {
277 return entry.color_;
278 }
279 }
280 return accent::kNone;
281}
